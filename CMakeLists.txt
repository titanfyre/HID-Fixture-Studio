cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(HIDFixtureStudio VERSION 0.1.2 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Use Visual Studio 2022 C++ tools for Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SDK_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/SDK" CACHE PATH "Vdevice SDK directory; override privately for the production edition")
option(ENABLE_ASAN "Enable AddressSanitizer for offline validation" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /PDBALTPATH:%_PDB%)
if(ENABLE_ASAN)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/DEBUG /INCREMENTAL:NO)
endif()
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
 set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_library(FixtureCore STATIC App/Fixture.cpp "${SDK_DIRECTORY}/Vdevice.cpp")
target_include_directories(FixtureCore PUBLIC App "${SDK_DIRECTORY}")
add_executable(HIDFixtureStudio WIN32 App/Main.cpp App/Session.cpp)
target_link_libraries(HIDFixtureStudio PRIVATE FixtureCore comctl32 comdlg32 shell32 advapi32 dwmapi gdiplus)
target_link_options(HIDFixtureStudio PRIVATE "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/App/App.manifest")
enable_testing()
add_executable(FixtureTests Tests/FixtureTests.cpp)
target_link_libraries(FixtureTests PRIVATE FixtureCore)
add_test(NAME FixtureModel COMMAND FixtureTests)
add_test(NAME OfflineApplication COMMAND HIDFixtureStudio --self-test)
add_executable(SessionTests Tests/SessionTests.cpp App/Session.cpp App/Fixture.cpp)
target_include_directories(SessionTests PRIVATE App "${SDK_DIRECTORY}")
add_test(NAME FakeVdeviceLifecycle COMMAND SessionTests)
